Code Example

Function to debounce API calls and improve performance:

function debounce<T extends (...args: any[]) => any>(
  func: T,
  delay: number
): (...args: Parameters<T>) => void {
  let timeoutId: ReturnType<typeof setTimeout> | null = null;
  
  return (...args: Parameters<T>) => {
    if (timeoutId) {
      clearTimeout(timeoutId);
    }
    
    timeoutId = setTimeout(() => {
      func(...args);
    }, delay);
  };
}

// Usage example
const searchAPI = debounce((query: string) => {
  fetch(`/api/search?q=${encodeURIComponent(query)}`)
    .then(response => response.json())
    .then(data => console.log(data));
}, 300);

Code Example #2

Solution to Codewars kata "Odd-numbered substrings" - Count how many odd-numbered substrings can be formed from a string of digits:

// Task: Given a string of integers, return the number of 
// odd-numbered substrings that can be formed.
// Example: "1341" -> substrings: 1, 1, 3, 13, 41, 341, 1341 = 7 numbers

function solve(s) {
  let count = 0;
  
  // Key insight: A number is odd if its last digit is odd
  // For each position with an odd digit, count all substrings 
  // ending at that position
  
  for (let i = 0; i < s.length; i++) {
    // If digit at position i is odd
    if (parseInt(s[i]) % 2 === 1) {
      // Number of substrings ending here = index + 1
      count += i + 1;
    }
  }
  
  return count;
}

// Test cases
console.log(solve("1341"));     // 7
console.log(solve("1357"));     // 10
console.log(solve("13471"));    // 12
